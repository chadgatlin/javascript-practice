let count = 0;

function cardCounter(card) {
 if (card === 2 || card === 3 || card === 4 || card === 5 || card === 6) {
   count += 1;
}
else if (card === 10 || card === "J" || card === "Q" || card === "K" || card === "A") {
  count -= 1;
}
if (count >= 1) {
  return count + " Bet";
} else {
  return count + " Hold";
}
}

console.log(cardCounter(9));
